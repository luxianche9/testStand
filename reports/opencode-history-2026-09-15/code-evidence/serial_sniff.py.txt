"""Passive CP-118U-I serial sniff: listen only, never transmit.

Captures whatever the DUT emits (boot banner and periodic telemetry) as raw hex
and text. Use it to tell a baud/clock problem (constant garbage) from a wiring
problem (silence) without the DUT needing to answer a command.
"""

from __future__ import annotations

import time

from hardwareharness import call as _raw_call
from hardwareharness import log, step

SERIAL = "PC1_MCP.CP118U_I_CARD_0"

T_PORT = "CP118U_I_MCP__cp118u_configure_port"
T_READ = "CP118U_I_MCP__cp118u_read_rx"
T_CLEAR = "CP118U_I_MCP__cp118u_clear_rx"

BAUDS = (2400, 4800, 9600, 19200, 38400, 57600, 115200, 230400)


def _unwrap(result: object) -> object:
    if isinstance(result, dict):
        structured = result.get("structuredContent")
        if isinstance(structured, dict):
            return structured.get("data", structured)
        if "data" in result:
            return result["data"]
    return result


def call(hardware_id: str, tool_name: str, arguments: dict, **kwargs: object) -> object:
    return _unwrap(_raw_call(hardware_id, tool_name, arguments, **kwargs))


def _printable_ratio(text: str) -> float:
    if not text:
        return 0.0
    good = sum(1 for ch in text if ch.isprintable() or ch in "\r\n\t")
    return good / len(text)


def main(inputs: dict[str, object]) -> dict[str, object]:
    channel = int(str(inputs.get("channel", 1)))
    seconds = float(str(inputs.get("seconds", 1.5)))

    results = []
    for baud in BAUDS:
        with step(f"{baud} baud 监听"):
            call(
                SERIAL,
                T_PORT,
                {
                    "channel": channel,
                    "baud": baud,
                    "data_bits": 8,
                    "parity": "N",
                    "stop_bits": 1,
                    "flow": 0,
                },
            )
            call(SERIAL, T_CLEAR, {"channel": channel})
            deadline = time.monotonic() + seconds
            while time.monotonic() < deadline:
                time.sleep(0.2)
            data = call(SERIAL, T_READ, {"channel": channel, "timeout_ms": 0, "max_bytes": 65536})
            text = str(data.get("text", ""))
            record = {
                "baud": baud,
                "bytes": int(data.get("bytes", 0)),
                "printable": round(_printable_ratio(text), 3),
                "hex": str(data.get("hex", ""))[:96],
                "sample": text.replace("\r", "\\r").replace("\n", "\\n")[:96],
            }
            results.append(record)
            log("监听结果", fields=record)

    best = max(results, key=lambda item: (item["printable"], item["bytes"]))
    log("可能匹配的波特率", fields={"baud": best["baud"], "printable": best["printable"]})
    return {"channel": channel, "results": results, "best": best}
